# Dependencies
# ---------------------
yum update -y
yum groupinstall -y "Development Tools"
yum install -y java-17-openjdk-devel
yum install -y maven

# Build
# ---------------------
git clone https://github.com/spring-projects/spring-petclinic.git
cd spring-petclinic
./mvnw spring-boot:run
